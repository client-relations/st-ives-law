## RULES (read before implementing)

1. **Every visible field is required.** If it renders on screen, the user must fill it before Next is enabled.
2. **Conditional fields** — if a field only appears because of a prior answer (e.g. urgency detail, postal address, prior relationship details), it is required the moment it appears.
3. **Dynamic list fields** — a count of 0 is acceptable (user has none). But if count > 0, every field inside every list item is required.
4. **Gateway Yes/No questions** precede all dynamic lists (properties, bank accounts, super, businesses, trusts, insurance). The gateway answer itself is required. If Yes, the list items' fields are all required. If No, no list fields appear.
5. **Back navigation is always allowed** — never block going backwards.
6. **Next button behaviour** — on click, run validation for the current page. If any errors exist: block navigation, show error banner at top of page listing all missing fields, highlight each failing field red with inline message, scroll to top. If no errors: clear errors and advance.
7. **Percentage totals** — residuary beneficiary percentages must sum to exactly 100% when count > 0. This is a hard block, not a warning.

    req(cData.c1HasExistingWill, "c1HasExistingWill", "Please indicate if Client 1 has an existing Will");
    if (cData.c1HasExistingWill === "Yes") {
      req(cData.c1WillDate, "c1WillDate", "Approximate date of existing Will is required");
      req(cData.c1WillKept, "c1WillKept", "Please indicate where the existing Will is kept");
    }
    if ((aData.engType || "").startsWith("Couple")) {
      req(cData.c2HasExistingWill, "c2HasExistingWill", "Please indicate if Client 2 has an existing Will");
      if (cData.c2HasExistingWill === "Yes") {
        req(cData.c2WillDate, "c2WillDate", "Approximate date of Client 2's existing Will is required");
        req(cData.c2WillKept, "c2WillKept", "Please indicate where Client 2's existing Will is kept");
      }
    }
  }

  // ─── PART D: EPA & Medical ───────────────────────────────────────────────
  // Build dynamic page titles using client preferred names
  const n1 = aData?.c1?.nick || aData?.c1?.first || "Client 1";
  const n2 = aData?.c2?.nick || aData?.c2?.first || "Client 2";

  if (pageTitle === `${n1} — EPA: Health & Personal`) {
    req(dData.c1EpaHealth, "c1EpaHealth", "Please indicate your EPA preference for health decisions");
    if ((dData.c1EpaHealth || "").startsWith("Yes")) {
      req(dData.c1HealthAttCount, "c1HealthAttCount", "Please enter the number of health attorneys");
      const count = parseInt(dData.c1HealthAttCount || 0);
      if (count === 0) {
        errors.push({ field: "c1HealthAttCount_min", message: "At least one health attorney must be nominated" });
      }
      reqListItem(dData.c1HealthAtts, count, "c1_health_att", [
        { key: "first",  label: "First name"        },
        { key: "last",   label: "Last name"         },
        { key: "rel",    label: "Relationship"      },
        { key: "email",  label: "Email"             },
        { key: "mobile", label: "Mobile"            },
        { key: "agreed", label: "Has agreed to act" },
      ]);
    }
  }

  if (pageTitle === `${n1} — Medical & End of Life`) {
    req(dData.c1Ahd,             "c1Ahd",             "Please indicate your Advance Health Directive preference");
    req(dData.c1LifeSustaining,  "c1LifeSustaining",  "Please indicate your life-sustaining treatment wishes");
    req(dData.c1OrganDonation,   "c1OrganDonation",   "Please indicate your organ donation wishes");
  }

  if (pageTitle === `${n1} — EPA: Financial`) {
    req(dData.c1EpaFin, "c1EpaFin", "Please indicate your EPA preference for financial decisions");
    if ((dData.c1EpaFin || "").startsWith("Yes")) {
      req(dData.c1EpaFinTrigger, "c1EpaFinTrigger", "Please indicate when the financial EPA takes effect");
      req(dData.c1FinAttCount,   "c1FinAttCount",   "Please enter the number of financial attorneys");
      const count = parseInt(dData.c1FinAttCount || 0);
      if (count === 0) {
        errors.push({ field: "c1FinAttCount_min", message: "At least one financial attorney must be nominated" });
      }
      reqListItem(dData.c1FinAtts, count, "c1_fin_att", [
        { key: "first",  label: "First name"        },
        { key: "last",   label: "Last name"         },
        { key: "rel",    label: "Relationship"      },
        { key: "email",  label: "Email"             },
        { key: "mobile", label: "Mobile"            },
        { key: "agreed", label: "Has agreed to act" },
      ]);
    }
  }

  if (pageTitle === `${n1} — Funeral & Legacy Wishes`) {
    req(dData.c1BurialPref,      "c1BurialPref",      "Please indicate burial or cremation preference");
    req(dData.c1FuneralReligion, "c1FuneralReligion", "Please describe religious or cultural observances (enter 'None' if not applicable)");
    req(dData.c1Legacy,          "c1Legacy",          "Please describe legacy wishes (enter 'None' if not applicable)");
  }

  // Repeat for C2 if couple — identical pattern with n2 / c2 prefix
  if ((aData.engType || "").startsWith("Couple")) {
    if (pageTitle === `${n2} — EPA: Health & Personal`) {
      req(dData.c2EpaHealth, "c2EpaHealth", "Please indicate Client 2's EPA preference for health decisions");
    }
    if (pageTitle === `${n2} — Medical & End of Life`) {
      req(dData.c2Ahd,            "c2Ahd",            "Please indicate Client 2's AHD preference");
      req(dData.c2LifeSustaining, "c2LifeSustaining",  "Please indicate Client 2's life-sustaining treatment wishes");
      req(dData.c2OrganDonation,  "c2OrganDonation",   "Please indicate Client 2's organ donation wishes");
    }
    if (pageTitle === `${n2} — EPA: Financial`) {
      req(dData.c2EpaFin, "c2EpaFin", "Please indicate Client 2's EPA preference for financial decisions");
    }
    if (pageTitle === `${n2} — Funeral & Legacy Wishes`) {
      req(dData.c2BurialPref,      "c2BurialPref",      "Burial preference required for Client 2");
      req(dData.c2FuneralReligion, "c2FuneralReligion", "Religious observances required for Client 2 (enter 'None' if not applicable)");
      req(dData.c2Legacy,          "c2Legacy",          "Legacy wishes required for Client 2 (enter 'None' if not applicable)");
    }
  }

  // ─── PART D: Additional Information ──────────────────────────────────────
  if (pageTitle === "Additional Information") {
    req(dData.c1Litigation,      "c1Litigation",      "Please indicate if Client 1 is involved in any litigation");
    req(dData.c1AdditionalInfo,  "c1AdditionalInfo",  "Please enter any additional information (enter 'None' if not applicable)");
  }

  // ─── PART D: Declaration ─────────────────────────────────────────────────
  if (pageTitle === "Declaration") {
    req(dData.c1SignDate, "c1SignDate", "Declaration date is required for Client 1");
    if ((aData.engType || "").startsWith("Couple")) {
      req(dData.c2SignDate, "c2SignDate", "Declaration date is required for Client 2");
    }
  }

  return errors;
}
